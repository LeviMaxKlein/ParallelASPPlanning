\section{Definition}
\begin{defi}[Multi-valued planning task]
    A STRIPS-like multivalued planning task according to \parencite{Helmert2006} is given by a 4 tuple $\langle \mathcal{F}, s_0, s_*, \mathcal{O} \rangle$, where:
    \begin{itemize}
      \item $\mathcal{F}$ is a finite set of state variables, also called \textbf{fluents}. Each $x \in \mathcal{F}$ has a finite domain $x^d$ and in every state, $x$ takes exactly one value from its domain.
      \item $s_0$, the \textbf{initial state}, is a total function, s.t. $s_0(x) \in x^d$ for each $x \in \mathcal{F}$
      \item $s_*$, the \textbf{goal conditions}, is a partial function, s.t. $s_*(x) \in x^d$, for each $x \in dom(s_*)$
      \item $\mathcal{O}$ is a finite set of operators, also called \textbf{actions}. We define an action as $a = \langle a^c, a^e \rangle$, where $a^c$ is the precondition of $a$ and $a^e$ is the postcondition of $a$. 
            Both $a^c$ and $a^e$ are partial variable assignments over $\mathcal{F}$.
            %TODO: genauer pre und postcondition erklären
    \end{itemize}
\end{defi}

\begin{example}
  Let us illustrate the definitions with a small example. The plan is to leave the house by opening the door, going outside, and finally closing the door.
  \begin{itemize}
    \item $\mathcal{F} = \{door, location\}$ with $door^d = \{open, closed\}$, $location^d = \{inside, outside\}$
    \item Initial state: $s_0 = \{door{=}closed, location{=}inside\}$
    \item Goal: $s_* = \{door{=}closed, location{=}outside\}$
    \item Actions: $\mathcal{O} = \{openDoor, closeDoor, goOut\}$, where
    \vspace{-1em}
    \begin{align*}
      openDoor &= \langle\{door{=}closed\}, \{door{=}open\}\rangle\\
      closeDoor &= \langle\{door{=}open\}, \{door{=}closed\}\rangle\\
      goOut &= \langle\{location{=}inside, door{=}open\}, \{location{=}outside\}\rangle
    \end{align*}
  \end{itemize}
\end{example}
\begin{defi}[Successor state]
  Let $s$ be state and $a \in \mathcal{O}$ an action.\\
  The \textit{successor state} $o(a,s)$, obtained by applying action $a$ in state $s$, is defined if the precondtion of $a$ holds, i.e., $a^c(x) = s(x)$ for all $x \in dom(a^c)$. Then
  $$
  s'(x) = o(a,s) = \begin{cases} a^e(x) & \text{if } x \in dom(a^e) \\ s(x) & \text{otherwise.} \end{cases}
  $$ 
\end{defi}
\begin{remark}
  Let $A_n = \langle a_1, \dots, a_n \rangle$ be a sequence of actions with length n. We define the application of $A_n$ to a state $s$ recursevily by
  $$
    o(A_n,s) = o(a_n,o(A_{n-1},s))
  $$
  if $o(A_i, s)$ is defined for all $1 \le i \le n$.
\end{remark}
\begin{defi}[Sequential plan]
  A \textit{sequential plan} is a sequence of actions $A_n = \langle a_1, \dots, a_n \rangle$, s.t. 
  $s' = o(A_n,s_0)$ is defined and $s'(x) = s_*(x)$ for all $x \in dom(s_*)$.
\end{defi}

\begin{example}
  Consider the planning task from the previous example. A sequential plan would be $A = \langle openDoor, goOut, closeDoor \rangle$. Since
  \begin{align*}
    o(A, s_0) &= o(\langle openDoor, goOut, closeDoor\rangle, \{door{=}closed, location{=}inside\})))\\
    &= o(\langle goOut, closeDoor \rangle, \{door{=}open, location{=}inside\}))\\
    &= o(closeDoor, \{door{=}open, location{=}outside\})\\
    &= \{door{=}closed, location{=}outside\} = s_*
  \end{align*}
\end{example}
\hspace{-1.5em}
To get to parallel actions and plans, we need the definition of confluent actions.

\begin{defi}[confluent]
  Let $A = \{a_1, \dots, a_n\} \subseteq \mathcal{O}$ be a set of actions.\\
  $A$ is \textit{confluent} $\logeq a_i^e(x) = a_j^e(x)$ for all $1 \le i < j \le n$ and each $x \in dom(a_i^e) \cap dom(a_j^e)$
\end{defi}

\begin{remark}
  $A$ is confluent $\Leftrightarrow B$ is confluent for all $B \subseteq A$
\end{remark}

\begin{example}
  In our example $\{openDoor, goOut\}$ and $\{closeDoor, goOut\}$ are confluent sets. \\
  We extend our example with the fluents $light$ and $key$ with $light^d = key^d = \{0, 1\}$ and two actions $lightsOut$ and $getKey$:
  \vspace{-1em}
  \begin{flalign*}
    lightsOut &= \langle\{location{=}inside, light{=}1\}, \{light{=}0\}\rangle& \\
    getKey &= \langle\{location{=}inside, key{=}0\}, \{key{=}1\}\rangle&
  \end{flalign*}
  \noindent Additionally, we modify the precondition: $openDoor^c = \{door{=}closed, key{=}1\}$.
  \noindent Our new goal is $s_* = \{location{=}outside, door{=}closed, light{=}0\}$. 
  
  \noindent Now for example $\{getKey, openDoor, goOut, lightsOut\}$ and $\{getKey,$ $closeDoor,$ $goOut,$ $lightsOut\}$ are also confluent sets.
\end{example}

\begin{defi}[Parallel representations]
  Let $s$ be a state and $A = \{a_1, \dots, a_n\}$ be a confluent set of actions. Then $A$ is
  \begin{enumerate}[label=(\roman*)]
    \item 
      \(
        \text{\textit{$\forall$-step serializable} in $s$}
        \begin{aligned}
          &\logeq\; o(\langle a_{\pi(1)}, \dots, a_{\pi(n)} \rangle, s)\ \text{is defined for all permutations }\pi
        \end{aligned}
      \)
    \item 
      \(
        \begin{aligned}[t]
          \text{\textit{$\exists$-step serializable} in $s$}
          &\logeq\; o(\langle a_{\pi(1)}, \dots, a_{\pi(n)} \rangle, s)\ \text{is defined for some permutation }\pi \\
          &\phantom{\logeq}\; ~\text{ and } a^c(x) = s(x) \text{ for each } a \in A \text{ and } x \in dom(a^c)
        \end{aligned}
      \)
    \item 
      \(
        \begin{aligned}[t]
          \text{\textit{relaxed $\exists$-step serializable} in $s$}
          &\logeq\; o(\langle a_{\pi(1)}, \dots, a_{\pi(n)} \rangle, s)\ \text{is defined for some} \\
          &\phantom{\logeq}\; ~\text{ permutation }\pi
        \end{aligned}
      \)
  \end{enumerate}
\end{defi}


\begin{remark}
  $A$ is $\forall$-step serializable $\Rightarrow$ $A$ is $\exists$-step serializable $\Rightarrow$ $A$ is relaxed $\exists$-step serializable.
\end{remark}

\begin{defi}[Parallel Plans]
  Let $A = \langle A_1, \dots, A_n \rangle$ be a sequence of confluent sets of actions. $A$ is a \textit{$\forall$-step, $\exists$-step or relaxed $\exists$-step plan} iff
  \begin{itemize}
    \item $s_n(x) = s_*(x)$ for each $x \in dom(s_*)$
    \item $A_i$ is $\forall$-step, $\exists$-step or relaxed $\exists$-step serializable in $s_{i-1}$ for $1 \le i \le n$, where \\
    $
    s_i(x) = \begin{cases} a^e(x) & \text{if } x \in dom(a^e) \\ s_{i-1}(x) & \text{else} \end{cases} \text{ for each } a \in A_i
    $
  \end{itemize}
\end{defi}

\begin{example}
  In our modified example we have:
  \begin{itemize}
    \item $\forall$-step plan: $\langle\{getKey, lightsOut\}, \{openDoor\}, \{goOut\}, \{closeDoor\}\rangle$
    \item $\exists$-step plan: $\langle\{getKey, lightsOut\}, \{openDoor\}, \{goOut, closeDoor\}\rangle$
    \item relaxed $\exists$-step plan: $\langle\{getKey, lightsOut, openDoor, goOut\}, \{closeDoor\}\rangle$
  \end{itemize}
\end{example}

\noindent Combining these facts with the right encodings, stable models corresponds to the \textit{sequential, $\forall$-step, $\exists$-step} or \textit{relaxed $\exists$-step plan}.

\begin{listing}[H]
  \caption{Core for sequential and parallel encodings for planning}
  \inputminted[frame=single,framesep=2mm,baselinestretch=1.2,linenos,lastline=9]{prolog}{encodings/EX_common.lp}
  \label{core}
\end{listing}
\noindent All planning algorithms share a common incremental encoding using the incmode for clingo as seen in Listing \ref{core} \parencite{clingo14}.
The program is divided into three subprograms: \textit{base}, \textit{step(t)} and \textit{check(t)}.
The \textit{base} subprogram contains the static part of the encoding, which is only grounded once. Line 2 defines the initial state $s_0$, 
where the predicate $holds(X,V,t)$ is used to represent fluents $X \in \mathcal{F}$ that have value $V\in dom(X)$ at timestep $t$. \\
\textit{check(t)} is a parametrized subprogram starting at $t=0$. The integrity constraint in Line 4 checks if $s_*(X)=s_t(X)$ $\forall X\in dom(s_*)$.
$Query(t)$ is true only for the latest integer $t$. \\
\textit{step(t)} is grounded after check fails for each timestep $t$ starting at $t=1$. Note that postconditions of applied actions in timestep $t$ are also set in $t$ and the preconditions have to hold in $t-1$.
Lines 6 defines holds for times step $t$, giving fluent $X$ one value $V$ from its domain and $occurs(A,t)$ in Line 7 are all possible actions in timestep $t$ obeying rules 8 and 9.
Line 8 ensures that the postcondition of an action is applied and also assert the confluence of any set of action applied in parallel since 
if $V = a^e(X) \ne \tilde{a}(X) = \tilde{V}$ either $holds(X,V,t)$ or $holds(X,\tilde{V}, t)$ would be false.
Line 9 makes sure that fluents not affected by any action keep their value from the previous timestep.
So far, the common core handles the transitions from one state to the next in regards of the postcondition of applied actions.

\begin{listing}[H]

\noindent Now for the extension for $\forall$-step plans as seen in Listing \ref{forall}, Line 1 is the same as in the sequential encoding.
As mentioned before, the integrity constraint in Line 8 of the common core already ensures confluence of applied actions. 
Since the sequence of the actions has to be valid for all permutations, we have to make sure that no action invalidates the precondition of any other action.
Let $a$ be an action applied in parallel and $X \in dom(a^c)$. There are two cases to consider: 
\begin{enumerate}
  \item the fluent $X$ remains unchanged, i.e. $X \in dom(a^c) \cap dom(a^e)$ and $a^c(X) = a^e(X)$ or $X \in dom(a^c)$ and $X \notin dom(a^e)$ 
  \item the postcondition of $a$ invalidates its precondition, i.e. $X \in dom(a^e) \cap a^c(X) \ne a^e(X)$ for an action a
\end{enumerate}
Line 2 handles the first case, while the second case is a little trickier. If there is such an action, there can not be another action $\tilde{a}$ with $X \in dom(\tilde{a}^e)$, 
since if $a^e(X)= \tilde{a}^e(X)$, the precondition of a is invalidatet and if $a^e(X) \ne \tilde{a^e}(X)$, the confluence is undermined. 
For this, we define these actions described in (ii) as $single(X,t)$ in Line 3 and the integrity constraint in Line 4 ensures that there is at most one such action for each fluent $X$ in timestep $t$.

\section{Proof of Correctness}
In order to prove the correctness of the presented encondings, we define the following:
\begin{itemize}
  \item $B$ is the rule in Line 1 of Listing \ref{seq}.
  \item $Q(t)$ is the integrity constraint in the check program directive (Line 5) of Listing \ref{core}.
  \item $S(t)$ are the rules and integrity constraint in the step program directive (Lines 6-9) of Listing \ref{core}.
  \item $S^\square(t)$ is the extension for the specific plans,
        where $\square = s$ for Listing \ref{seq}, $\square=\forall$ for Listing \ref{forall}, $\square=\exists$ for Listing \ref{exists},
        and $\square=R$ for Listing \ref{relaxed_exists}.  
\end{itemize}
For the next theorems, let $I$ be the set of facts representing a planning task $\langle \mathcal{F}, s_0, s_*, \mathcal{O} \rangle$,
  $\langle a_1, \dots, a_n \rangle$ be a sequence of actions and $\langle A_1, \dots, A_m \rangle$ be a sequence of sets of actions.
\begin{theorem}[Correctness of sequential plans]
  \phantom{}\\
  $\langle a_1, \dots, a_n \rangle$ is a sequential plan iff
  \[
    I \cup B \cup Q(0) \cup \bigcup_{t=1}^{n} \left( Q(t) \cup S(t) \cup S^s(t) \right) \cup \{query(n). \} 
  \]
  has a stable model $M$, s.t.$\{\langle a, t \rangle | occurs(a,t) \in M \} = \{\langle a_t, t \rangle | 1 \le t \le n \}$.
\end{theorem}

\begin{proof}
  Let $P^s(n) = I \cup B \cup Q(0) \cup \bigcup_{t=1}^{n} \left( Q(t) \cup S(t) \cup S^s(t) \right) \cup \{query(n). \}$. \\
  ($\Rightarrow$) $\langle a_1, \dots, a_n \rangle$ is a sequential plan $\Rightarrow$ any stable model $M$ of $P^s(n)$ s.t. $\{\langle a, t \rangle | occurs(a,t) \in M \} = \{\langle a_t, t \rangle | 1 \le t \le n \}$
  the rule in B ($holds(X, V, 0) = s_0=(X)$), the choice rule in $S(t)$ (for given X and t, there is only one V for holds(X, V, t)) and both integrity constraints in $S(i)$ ($holds(X,V,t) = a^e(X)$ if $X \in dom(a^e)$, $s_{t-1}(X)$ otherwise) make sure that
  $holds(X, V, 0) = s_0(X)$ and $holds(X, V, t) = o(a_t, s_{t-1})(X)$, i.e.
  \[
    \{\langle X,V,t\rangle | holds(X,V,t) \in M\} = \{\langle X,V,t\rangle | X \in \mathcal F, v = o(\langle a_1, \dots, a_t\rangle, s_0)\}.
  \]
\end{proof}
