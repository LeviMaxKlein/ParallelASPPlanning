\section{Definition}
\begin{defi}[Multi-valued planning task]
    A STRIPS-like multivalued planning task according to \parencite{Helmert2006} is given by a 4 tuple $\langle \mathcal{F}, s_0, s_*, \mathcal{O} \rangle$, where:
    \begin{itemize}
      \item $\mathcal{F}$ is a finite set of state variables, also called \textbf{fluents}. Each $x \in \mathcal{F}$ has a finite domain and in every state, $x$ takes exactly one value from its domain.
      \item $s_0$, the \textbf{initial state}, is a total function,, s.t. $s_0(x) \in dom(x)$ for each $x \in \mathcal{F}$, i.e. a function that maps each fluent to a value from its domain.
      \item $s_*$, the \textbf{goal conditions}, is a partial function, s.t. $s_*(x) \in dom(x)$, for each $x \in dom(s_*)$ i.e. a function that specifies the desired values for a subset of fluents.
      \item $\mathcal{O}$ is a finite set of operators, also called \textbf{actions}. We define an action $a$ as $\langle a^c, a^e \rangle$, where $a^c$ is the precondition and $a^e$ is the postcondition of $a$. 
            Both $a^c$ and $a^e$ are partial functions, s.t. $a^c(x) \in dom(x)$ for each $x \in dom(a^c)$ and $a^e(x) \in dom(x)$ for each $x \in dom(a^e)$.
    \end{itemize}
\begin{remark}
  An action $a$ also has a finite set of effects $eff$, where $eff = \langle cond, x, v \rangle$. 
  $cond$, the effect condition, is a (possibly empty) partial function, s.t. $cond(x) \in dom(x)$ for each $x \in dom(cond)$. 
  $x \in \mathcal F$ is the effected fluent and $v \in dom(x)$ is the value assigned to $x$ if $cond$ holds.
\end{remark}
\end{defi}

\begin{example}
  Let us illustrate the definitions with a small example. The plan is to leave the house by opening the door, going outside, and finally closing the door.
  \begin{itemize}
    \item $\mathcal{F} = \{door, location\}$ with $dom(door) {=} \{open, closed\}$, $dom(location) {=} \{inside, outside\}$
    \item Initial state: $s_0 = \{door{=}closed, location{=}inside\}$
    \item Goal: $s_* = \{door{=}closed, location{=}outside\}$
    \item Actions: $\mathcal{O} = \{openDoor, closeDoor, goOut\}$, where
    \vspace{-1em}
    \begin{align*}
      openDoor &= \langle\{door{=}closed\}, \{door{=}open\}\rangle\\
      closeDoor &= \langle\{door{=}open\}, \{door{=}closed\}\rangle\\
      goOut &= \langle\{location{=}inside, door{=}open\}, \{location{=}outside\}\rangle
    \end{align*}
  \end{itemize}
\end{example}
\begin{defi}[Successor state]
  Let $s$ be state and $a \in \mathcal{O}$ an action.\\
  The \textit{successor state} $o(a,s)$, obtained by applying action $a$ in state $s$, is defined if the precondtion of $a$ holds, i.e., $a^c(x) = s(x)$ for all $x \in dom(a^c)$. Then
  $$
  s'(x) = o(a,s) = \begin{cases} a^e(x) & \text{if } x \in dom(a^e) \\ s(x) & \text{otherwise.} \end{cases}
  $$ 
\end{defi}
\begin{remark}
  Let $A_n = \langle a_1, \dots, a_n \rangle$ be a sequence of actions with length n. We define the application of $A_n$ to a state $s$ recursevily by
  $$
    o(A_n,s) = o(a_n,o(A_{n-1},s))
  $$
  if $o(A_i, s)$ is defined for all $1 \le i \le n$.
\end{remark}
\begin{defi}[Sequential plan]
  A \textit{sequential plan} is a sequence of actions $A_n = \langle a_1, \dots, a_n \rangle$, s.t. 
  $s' = o(A_n,s_0)$ is defined and $s'(x) = s_*(x)$ for all $x \in dom(s_*)$.
\end{defi}

\begin{example}
  Consider the planning task from the previous example. One possible sequential plan is $A = \langle openDoor, goOut, closeDoor \rangle$. Since
  \begin{align*}
    o(A, s_0) &= o(\langle openDoor, goOut, closeDoor\rangle, \{door{=}closed, location{=}inside\})))\\
    &= o(\langle goOut, closeDoor \rangle, \{door{=}open, location{=}inside\}))\\
    &= o(closeDoor, \{door{=}open, location{=}outside\})\\
    &= \{door{=}closed, location{=}outside\} = s_*
  \end{align*}
\end{example}
\hspace{-1.5em}

\noindent For parallel actions and plans, we first need the definition of a confluent set of actions.

\begin{defi}[confluent]
  Let $A = \{a_1, \dots, a_n\} \subseteq \mathcal{O}$ be a set of actions.\\
  $A$ is \textit{confluent} $\logeq a_i^e(x) = a_j^e(x)$ for all $1 \le i < j \le n$ and each $x \in dom(a_i^e) \cap dom(a_j^e)$
\end{defi}

\begin{remark}
  $A$ is confluent $\Leftrightarrow B$ is confluent for all $B \subseteq A$
\end{remark}

\begin{example}
  In our example $\{openDoor, goOut\}$ and $\{closeDoor, goOut\}$ are confluent sets. \\
  We extend the example with the fluents $light$ and $key$ where $dom(light) = dom(key) = \{0, 1\}$ and two actions $lightsOut$ and $getKey$:
  \vspace{-1em}
  \begin{align*}
    lightsOut &= \langle\{location{=}inside, light{=}1\}, \{light{=}0\}\rangle \\
    getKey &= \langle\{location{=}inside, key{=}0\}, \{key{=}1\}\rangle
  \end{align*}
  \noindent Additionally, we modify the precondition: $openDoor^c = \{door{=}closed, key{=}1\}$
  and our new goal is $s_* = \{location{=}outside, door{=}closed, light{=}0\}$. \\
  Now for example $\{getKey, openDoor, goOut, lightsOut\}$ and $\{getKey,$ $closeDoor,$ $goOut,$ $lightsOut\}$ are also confluent sets of actions.
\end{example}

\begin{defi}[Parallel representations]
  Let $s$ be a state and $A = \{a_1, \dots, a_n\}$ be a confluent set of actions. Then $A$ is
  \begin{enumerate}[label=(\roman*)]
    \item 
      \(
        \text{\textit{$\forall$-step serializable} in $s$}
        \begin{aligned}
          &\logeq\; o(\langle a_{\pi(1)}, \dots, a_{\pi(n)} \rangle, s)\ \text{is defined for all permutations }\pi
        \end{aligned}
      \)
    \item 
      \(
        \begin{aligned}[t]
          \text{\textit{$\exists$-step serializable} in $s$}
          &\logeq\; o(\langle a_{\pi(1)}, \dots, a_{\pi(n)} \rangle, s)\ \text{is defined for some permutation }\pi \\
          &\phantom{\logeq}\; ~\text{ and } a^c(x) = s(x) \text{ for each } a \in A \text{ and } x \in dom(a^c)
        \end{aligned}
      \)
    \item 
      \(
        \begin{aligned}[t]
          \text{\textit{relaxed $\exists$-step serializable} in $s$}
          &\logeq\; o(\langle a_{\pi(1)}, \dots, a_{\pi(n)} \rangle, s)\ \text{is defined for some} \\
          &\phantom{\logeq}\; ~\text{ permutation }\pi
        \end{aligned}
      \)
  \end{enumerate}
\end{defi}

\begin{remark}
  $A$ is $\forall$-step serializable $\Rightarrow$ $A$ is $\exists$-step serializable $\Rightarrow$ $A$ is relaxed $\exists$-step serializable.
\end{remark}

\begin{defi}[Parallel Plans]
  Let $A = \langle A_1, \dots, A_n \rangle$ be a sequence of confluent sets of actions. $A$ is a \textit{$\forall$-step, $\exists$-step or relaxed $\exists$-step plan} iff
  \begin{itemize}
    \item $s_n(x) = s_*(x)$ for each $x \in dom(s_*)$
    \item $A_i$ is $\forall$-step, $\exists$-step or relaxed $\exists$-step serializable in $s_{i-1}$ for $1 \le i \le n$, where \\
    $
    s_i(x) = \begin{cases} a^e(x) & \text{if } x \in dom(a^e) \\ s_{i-1}(x) & \text{else} \end{cases} \text{ for each } a \in A_i
    $
  \end{itemize}
\end{defi}

\begin{example}
  In our modified example we have:
  \begin{itemize}
    \item $\forall$-step plan: $\langle\{getKey, lightsOut\}, \{openDoor\}, \{goOut\}, \{closeDoor\}\rangle$
    \item $\exists$-step plan: $\langle\{getKey, lightsOut\}, \{openDoor\}, \{goOut, closeDoor\}\rangle$
    \item relaxed $\exists$-step plan: $\langle\{getKey, lightsOut, openDoor, goOut\}, \{closeDoor\}\rangle$
  \end{itemize}
\end{example}

\noindent Combining these facts with the right encodings, stable models correspond to the \textit{sequential, $\forall$-step, $\exists$-step} or \textit{relaxed $\exists$-step plans}.

\begin{listing}[H]
  \caption{Core for sequential and parallel encodings for planning}
  \inputminted[frame=single,framesep=2mm,baselinestretch=1.2,linenos,lastline=9]{prolog}{encodings/EX_common.lp}
  \label{core}
\end{listing}
\noindent All planning algorithms share a common incremental encoding using the incmode for clingo as seen in Listing \ref{core} \parencite{clingo14}.
The program is divided into three subprograms: \textit{base}, \textit{step(t)} and \textit{check(t)}. \\
The \textit{base} subprogram contains the static part of the encoding, which is only grounded once. 
Line 2 defines the initial state $s_0$, where the predicate $holds(X,V,t)$ is used to represent fluents $X \in \mathcal{F}$ that have value $V\in dom(X)$ at timestep $t$. \\
\textit{check(t)} is a parametrized subprogram starting at $t=0$. The integrity constraint in Line 4 checks if $s_*(X)=s_t(X)$ $\forall X\in dom(s_*)$.
$Query(t)$ is true only for the latest integer $t$. \\
\textit{step(t)} is grounded after check fails for each timestep $t$ starting at $t=1$. Note that postconditions of applied actions in timestep $t$ are also set in $t$ and the preconditions have to hold in $t-1$.
Lines 6 defines holds for times step $t$, giving fluent $X$ one value $V$ from its domain and $occurs(A,t)$ in Line 7 are all possible actions in timestep $t$ obeying rules 8 and 9.
Line 8 ensures that the postcondition of an action is applied and also assert the confluence of any set of action applied in parallel since 
if $V = a^e(X) \ne \tilde{a}(X) = \tilde{V}$ either $holds(X,V,t)$ or $holds(X,\tilde{V}, t)$ would be false.
Line 9 makes sure that fluents not affected by any action keep their value from the previous timestep.
So far, the common core handles the transitions from one state to the next in regards of the postcondition of applied actions.

\begin{listing}[H]

\noindent Now for the extension for $\forall$-step plans as seen in Listing \ref{forall}, Line 1 is the same as in the sequential encoding.
As mentioned before, the integrity constraint in Line 8 of the common core already ensures confluence of applied actions. 
Since the sequence of the actions has to be valid for all permutations, we have to make sure that no action invalidates the precondition of any other action.
Let $a$ be an action applied in parallel and $X \in dom(a^c)$. There are two cases to consider: 
\begin{enumerate}
  \item the fluent $X$ remains unchanged, i.e. $X \in dom(a^c) \cap dom(a^e)$ and $a^c(X) = a^e(X)$ or $X \in dom(a^c)$ and $X \notin dom(a^e)$ 
  \item the postcondition of $a$ invalidates its precondition, i.e. $X \in dom(a^e) \cap a^c(X) \ne a^e(X)$ for an action a
\end{enumerate}
Line 2 handles the first case, while the second case is a little trickier. If there is such an action, there can not be another action $\tilde{a}$ with $X \in dom(\tilde{a}^e)$, 
since if $a^e(X)= \tilde{a}^e(X)$, the precondition of a is invalidatet and if $a^e(X) \ne \tilde{a^e}(X)$, the confluence is undermined. 
For this, we define these actions described in (ii) as $single(X,t)$ in Line 3 and the integrity constraint in Line 4 ensures that there is at most one such action for each fluent $X$ in timestep $t$.

\section{Proof of Correctness}
In order to prove the correctness of the presented encondings, we define the following:
\begin{itemize}
  \item $B$ is the rule in Line 1 of Listing \ref{seq}.
  \item $Q(t)$ is the integrity constraint in the check program directive (Line 5) of Listing \ref{core}.
  \item $S(t)$ are the rules and integrity constraint in the step program directive (Lines 6-9) of Listing \ref{core}.
  \item $S^p(t)$ is the extension for the specific plans,
        where $p = s$ for Listing \ref{seq}, $p=\forall$ for Listing \ref{forall}, $p=\exists$ for Listing \ref{exists},
        and $p=R$ for Listing \ref{relaxed_exists}.
  \item $I$ is the set of facts representing a planning task $\langle \mathcal{F}, s_0, s_*, \mathcal{O} \rangle$.
  \item $\langle a_1, \dots, a_n \rangle$ is a sequence of actions.
  \item $\langle A_1, \dots, A_m \rangle$ is a a sequence of sets of actions.
  \item $P^p(n) := I \cup B \cup Q(0) \cup \bigcup_{t=1}^{n} \left( Q(t) \cup S(t) \cup S^p(t) \right) \cup \{query(n). \}$, where $p \in \{s, \forall, \exists, R\}$.
\end{itemize}

\begin{theorem}[Correctness of sequential plans \parencite{DIMOPOULOS_2019}]
  \phantom{}\\
  $\langle a_1, \dots, a_n \rangle$ is a sequential plan $\iff$ $P^s(n)$ has a stable model $M$, s.t. \\
  $\{\langle a, t \rangle ~|~ occurs(a,t) \in M \} = \{\langle a_t, t \rangle ~|~ 1 \le t \le n \}$.
\end{theorem}

\begin{proof}
  ($\Rightarrow$): Let $\langle a_1, \dots, a_n \rangle$ be a sequential plan. \\
  First we show that for any stable model $M$ of $P^s(n)$ s.t. $\{\langle a, t \rangle | occurs(a,t) \in M \} = \{\langle a_t, t \rangle | 1 \le t \le n \}$, the rules and integrity constraints ensures that 
  $\{\langle X, V, t \rangle ~|~ holds(X, V, t) \in M \} = \{\langle X, V, t \rangle ~|~ X \in \mathcal{F}, V = o(\langle a_1, \dots, a_t\rangle, s_0), 0 \le t \le n \}$.
  \begin{enumerate}[label=(\roman*)]
    \item the rule in B ensures that $holds(X, V, 0) = s_0(X)$ forall $X \in \mathcal F$
    \item for any $1 \le t \le n$, both integrity constraints in S(t) ensure that $holds(X,V,t) = a_t^e(X)$ if $X \in dom(a_t^e)$ and $holds(X,V,t) = s_{t-1}(X)$ otherwise.
  \end{enumerate}
  Therefore, the only stable model $M$ that $P^s(n)$ can have is
  \begin{align*}
        M = I &\cup \{occurs(a_t, t) ~|~ 1 \le t \le n \} \\
              &\cup \{holds(X, V, t) ~|~ X \in \mathcal F, v = o(\langle a_1, \dots, a_t\rangle, s_0), 0 \le t \le n\} \cup \{query(n)\}.
  \end{align*}
  s.t. $\{\langle a, t \rangle ~|~ occurs(a,t) \in M \} = \{\langle a_t, t \rangle ~|~ 1 \le t \le n \}$. \\
  One can easily check that all rules and integrity constraints in $P^s(n)$ are satisfied and M is indeed a stable model of $P^s(n)$.\\
  ($\Leftarrow$): Let $M$ be a stable model of $P^s(n)$ s.t. $\{\langle a, t \rangle ~|~ occurs(a,t) \in M \} = \{\langle a_t, t \rangle ~|~ 1 \le t \le n \}$.
  We proof by induction over t, that for all $1 \le t \le n$:
  \[
      \{\langle X, V \rangle ~|~ holds(X,V,t-1)\in M \} = \{\langle X, V \rangle ~|~ X \in \mathcal F, V=o(\langle a_1, \dots, a_{t-1}\rangle, s_0)(X)\}.
  \]
  \textit{I-Start: t=0} \\
  The Rule in B implies: 
  $\{\langle X, V \rangle ~|~ holds(X, V, 0) \in M \} = \{\langle X, V \rangle ~|~ X \in \mathcal F, s_0(X) = V\}$. \\
  \textit{I-Step:} $t \mapsto t+1$ \\
  The integrity constraint in Line 1 of Listing \ref{seq} in $S^s(t)$ ensures that the successor state of $s(t)$, i.e. $o(a_{t+1}, o(\langle a_1, \dots, a_{t}\rangle, s_0))$ is defined.
  Additionally the choice rule in Line 6 of Listing \ref{core} and the integrity constaints in $S(t)$ guarentee that
  \[
    \{\langle X, V \rangle ~|~ holds(X,V,t+1) \in M\} = \{\langle X, V \rangle ~|~ X \in \mathcal F, V = o(\langle a_1, \dots, a_{t+1} \rangle, s_0)(X)\}.
  \]
  Finally, with the fact $query(n)$, the integrity constraint in $Q(n)$ yields that $o(\langle a_1, \dots, a_n \rangle, s_0)(X)$ $=$ $s_*(X)$ for all $X \in dom(s_*)$.
  Thus, $\langle a_1, \dots, a_n \rangle$ is a sequential plan.
\end{proof}

\begin{theorem}[Correctness of parallel plans \parencite{DIMOPOULOS_2019}]
  \phantom{} \\
  $\langle A_1, \dots, A_m \rangle$ is a $\forall$-/$exists$-/relaxed $exists$-step plan $\iff P^p(m)$ has a stable model M s.t., \\
  $\{\langle a, t \rangle ~|~ occurs(a,t) \in M \} = \{\langle a, t \rangle ~|~ a \in A_t, 1 \le t \le m \}$ with $p \in \{\forall, \exists, R\}$.
\end{theorem}

\begin{proof}
  ($\Rightarrow$) $\langle A_1, \dots, A_m \rangle$ is a $\forall$-/$exists$-/\textit{relaxed} $exists$-step plan. \\
  For each $1 \le t \le m$, let $s_t$ be the state s.t. $s_t(X) = \begin{cases} a^e(X) & \forall a \in A_t \text{ and } X \in dom(a^e) \\ s_{t-1}(X) & \forall X \in \mathcal F \setminus \bigcup_{a \in A_t} dom(a^e) \end{cases}$.
  Then, as in the proof above, for any stable model $M$ of $P^p(m)$ s.t. $\{\langle a, t \rangle ~|~ occurs(a,t) \in M\} = \{\langle a,t \rangle ~|~ a \in A_t, 1 \le t \le m \}$, 
  the rule in $B$ together with the choice rule and the integrity constraints in $S(t)$ make sure that
  $\{\langle X,V,t\rangle ~|~ holds(X,V,t) \in M\} = \{\langle X,V,t\rangle ~|~ X \in \mathcal F, V = s_t(X), 0 \le t \le m\}$.
  We now split the proof into the different cases: \\

  \noindent \textbf{Case 1: }$\langle A_1, \dots, A_m \rangle$ is a $\forall-$step plan. \\
  The atom $single(X,t)$ is derived by the rule in $S^\forall(t)$ iff $\exists a \in A_i$ s.t. $X \in dom(a^c) \cap dom(a^e)$ and $a^c(X) \ne a^e(X)$ for $1 \le t \le m$.
  Hence $P^\forall(m)$ cannot have any stable model $M$ apart from
  \begin{align*}
    M = I &\cup \{occurs(a,t) ~|~ a \in A_t, 1 \le t \le m\} \\
          &\cup \{holds(X,V,t) ~|~ X \in \mathcal F, V = s_t(X), 0 \le t \le m\} \\
          &\cup \{single(X,t) ~|~ X \in dom(a^c) \cap dom(a^e), a^c(X) \ne a^e(X), a \in A_t, 1 \le t \le m \} \\
          &\cup \{query(m).\}
  \end{align*}
  s.t. $\{\langle a, t \rangle ~|~ occurs(a,t) \in M\} = \{\langle a, t \rangle ~|~ a \in A_t, 1 \le t \le m\}$.
  Since for $t \in \{1, \dots, m \}$, the successor state $s_t= o(\langle a_1, \dots, a_k \rangle, s_{t-1})$ is defined for any sequence $\langle a_1, \dots, a_k \rangle$ s.t. $\{a_1, \dots, a_k\} = A_t$,
  we have that M satisfies all integrity constraints in $S^\forall(t)$. One can easily check that all other rules and integrity constraints in $P^\forall(m)$ are satisfied as well.
  Therefore M is a stable model of $P^\forall(m)$. \\

  \textbf{Case 2: }$\langle A_1, \dots, A_m \rangle$ is a $\exists-$step plan. \\
\end{proof}

To test and compare t he different parallel planning encodings against each other and against the sequential planning encoding, we set up a python pipeline using the lab framework \parencite{seipp-et-al-zenodo2017}
and the unofficial collection of (sequential) IPC benchmark instances from the downward project \footnote{\url{https://github.com/aibasel/downward-benchmarks}}. \\
Starting at a planning task in the Planning Domain Definition Language (PDDL), we first preprocess it using cpddl \parencite{fiser_2025_16423302} that converts it into a finite domain representation (FDR).
Plasp then takes the FDR as input and translates it into ASP facts representing the planning task. Further information of this step can be found in \parencite{DIMOPOULOS_2019}.
Using clingo with a modified version of the incremental encodings presented above together with the facts from the output of plasp, we can then compute sequential, $\forall$-step, $\exists$-step or relaxed $\exists$-step plans as stable models.
Finally, to validate the given plan, we postprocess the parallel representation back into a sequential plan, using the postprocess encoding provided by plasp and validate the plan with VAL \footnote{\url{https://github.com/KCL-Planning/VAL}}.
Each step has some optimizations and configurations that will be explained in the following sections.

\subsection{Mutexes}
\begin{defi}[mutex and mutex groups]
  A set of facts $M \subseteq \mathcal F$ is called 
  \begin{enumerate}[label=(\roman*)]
    \item a \textit{mutex} $~~~~~~~\logeq$ for all reachable states $s: M \not\subseteq s$.
    \item a \textit{mutex group} $\logeq$ for all reachable states $s: | M \cap s | \le 1$.
  \end{enumerate}
\end{defi}
\noindent In other words, a mutex is a set of fluents that can never be true at the same time in any reachable state,
while a mutex group is a set of fluents where at most one fluent can be true at the same time.
Mutexes and mutex groups can be used to optimize the planning encodings by reducing the search space.
For this, we extend our endcoding in Listing \ref{core} with the following rules:
\begin{listing}[H]
  \caption{Expansion for mutexes}
  \begin{minted}[frame=single,framesep=2mm,baselinestretch=1.2,linenos]{prolog}
    #program base.
    ...
    mutex(G,X) :- mutexGroup(G), contains(G,X,V), fluent(X,V).
    mutex(G)   :- mutexGroup(G), #count{X : mutex(G,X)} > 1.
    :- mutex(G), #count{X,V : holds(X,V,0), contains(G,X,V)} > 1.^
    ...
    #program step(t).
    ...
    :- mutex(G), #count{X,V : holds(X,V,t), contains(G,X,V)} > 1.
    ...
  \end{minted}
  \label{mutex}
\end{listing}
\noindent Here, the mutex group G is given by plasp as facts of the form $mutexGroup/1$ and $contains/3$.
In the base program, we first define which fluents belong to a mutex group (Line 3), where a group must contain at least two fluents (Line 4).
The integrity constraint in Line 5 ensures that no more than one fluent of each mutex group holds in the initial state and the integrity constraint in Line 9 ensures the same for each state in timestep t.

\subsection{Guess and Check}
An optimization specific to the (relaxed) $\exists$-step encodings is the guess and check approach. 
The general idea is to first guess a solution without looking for actions that circularly invalidate each other and afterwards check if the guessed solution has a cycle.
Formally, we encode the problem by a pair of programs $\langle G, C \rangle$ where $G$ is the guessing program and $C$ is the checking program. 
Stable models $M$ of $G$ yields a solution if $C \cap M$ is $unsatisfiable$. 
For making "systematic" guesses, \parencite{DIMOPOULOS_2019} proposes to use the core program ($S(t)$) in Listing \ref{core} 
together with the integrity constraint in Line 1 of Listing \ref{seq}, \ref{forall} or \ref{exists} ($B$) and the facts representing a planning task.
Thus, a stable model $M$ of $G$ is s.t. the pre- and postconditons hold for all actions applied in parallel (Line 8 of Listing \ref{core} and Line 1 of Listing \ref{seq}, \ref{forall} or \ref{exists}).
and the integrity constraint in Line 8 of Listing \ref{core} also ensures confluence of applied actions. 
The only thing remained to be checked is if actions circularly invalidate each other, i.e. $a^e(X) \ne \tilde a^c(X)$ and $\tilde a^e(\tilde X) \ne \tilde a^c(\tilde X)$.
This can be accomplished by taking the atoms over $occurs/2$ from $M$ as facts together with a program $C$ containing the facts representing a planning task
and the following encoding:
\begin{listing}[H]
  \caption{Encoding for checking circular invalidation}
  \inputminted[frame=single,framesep=2mm,baselinestretch=1.2,linenos]{prolog}{encodings/EX_check.lp}
  \label{check}
\end{listing}
\noindent Here, the maximum value for $t$ in subprogram $check$ and $step$ is $m = max \{t ~|~ occurs(a, t)\}$ and
$C$ becomes $unsatisfiable$ if $t > m$.
Like in Listing \ref{exists}, we check for each $t$ if all actions $A \in occurs(A,t)$ can be applied by the predicate $ready$ (Line 7-11).
If actions in $A$ circularly invalidate each other, $ready(A,t)$ is underivable for these actions and thus $cycle(t)$ (Line 12) in step $t$ is derived.
Hence, $C$ becomes satisfiable if there is a circular invalidation in step $t$, which means that there is no permutation of the actions A, s.t. the successor state is defined 
and therefore not (relaxed) $\exists$-step serializable.
